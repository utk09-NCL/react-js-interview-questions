export function Q4B() {
  const [count, setCount] = useState(0);
  console.log("Q4B before:", count);

  if (count === 0) {
    setCount(1);
  }

  console.log("Q4B after:", count);
  return (
    <>
      <p>Count: {count}</p>
      <button className="primary" onClick={() => setCount(prev => prev + 1)}>+1</button>
    </>
  );
}
